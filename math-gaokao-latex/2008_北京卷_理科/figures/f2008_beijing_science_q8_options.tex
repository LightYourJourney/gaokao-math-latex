\begin{tikzpicture}[semithick, every node/.style={inner sep=1.2pt}]

  \begin{scope}[shift={(2.4,0)}]
    \draw[->] (0,0) -- (1.6,0) node[right,font=\scriptsize] {$x$};
    \draw[->] (0,0) -- (0,1.2) node[above,font=\scriptsize] {$y$};

  \end{scope}
