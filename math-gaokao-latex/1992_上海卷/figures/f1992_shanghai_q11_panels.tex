\begin{tikzpicture}[line width=.45pt,x=.85cm,y=.85cm,font=\small]
  \foreach \panel/\shift/\growth/\opening in {A/0/1/1,B/3.3/-1/1,C/6.6/1/-1,D/9.9/-1/-1}{
    \begin{scope}[xshift=\shift cm]
      \draw[->] (-1.3,0)--(1.4,0) node[right] {$x$};
      \draw[->] (0,-1.35)--(0,1.65) node[above] {$y$};
      \draw[domain=-1.2:1.2,samples=65,smooth] plot (\x,{.6*pow(2,\growth*\x)});
      \draw[domain=-1.2:1.2,samples=65,smooth] plot (\x,{\opening*.3*\x*\x});
      \draw (-.045,.6)--(.045,.6);
      \node[left,inner sep=1pt,xshift=-2pt,yshift=5pt] at (0,.6) {$1$};
      \ifnum\opening=1
        \draw (1,-.045)--(1,.045);
        \node[below,inner sep=1pt] at (1,0) {$1$};
      \fi
      \node[below left,inner sep=1pt] at (0,0) {$O$};
      \node at (0,-1.8) {（\panel）};
    \end{scope}
  }
\end{tikzpicture}
