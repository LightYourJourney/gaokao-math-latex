\begin{tikzpicture}[x=1.05cm, y=1.15cm, font=\footnotesize, >=Stealth,

  % 坐标轴（原点处断开）

  \node[right] at (6.62,0.28) {月用电量/度};
  \draw[->] (0,0) -- (0,3.75) node[above] {$\dfrac{\text{频率}}{\text{组距}}$};

  \draw (-0.30,0.06) -- (-0.22,-0.06) -- (-0.14,0.06) -- (-0.06,-0.06);

  % 虚线参考高度
  \foreach \y/\lab in {3.0/0.0060, 2.2/x, 1.8/0.0036, 1.2/0.0024, 0.6/0.0012} {

  }

  % 六个矩形
  \draw (0,0) rectangle (1,1.2);
  \draw (1,0) rectangle (2,1.8);
  \draw (2,0) rectangle (3,3.0);

  \draw (5,0) rectangle (6,0.6);

  \node[left] at (-0.08,1.8) {$0.0036$};
  \node[left] at (-0.08,1.2) {$0.0024$};
  \node[left] at (-0.08,0.6) {$0.0012$};
  \node[left] at (-0.62,0) {$0$};

  % 横轴刻度
  \foreach \i/\lab in {0/50, 1/100, 2/150, 3/200, 4/250, 5/300, 6/350} {
    \draw (\i,0.05) -- (\i,-0.05);
    \node[below] at (\i,-0.08) {$\lab$};
  }

\end{tikzpicture}
