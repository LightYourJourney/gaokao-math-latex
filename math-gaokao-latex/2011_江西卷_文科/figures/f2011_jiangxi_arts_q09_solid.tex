\begin{tikzpicture}[>=stealth, semithick, font=\small, scale=0.85]
  % 几何体
  \coordinate (A) at (0,0); \coordinate (B) at (3,0);

  \end{scope}

    \draw[thick] (0,0)--(1.4,1.4);
    \node at (0.7,-0.3) {(D)};
  \end{scope}
