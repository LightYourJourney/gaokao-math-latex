% 原页：第 2 页第 1 栏，书页码 680。示意，p=PA。
\begin{tikzpicture}[>=stealth, semithick, font=\small, scale=0.95]
  \coordinate (B) at (0,0);
  \coordinate (C) at (4,0);
  \coordinate (P) at (0,2.2);
  \coordinate (D) at (1.8,2.2);
  \coordinate (A) at (0,4.0);
  \coordinate (Ap) at (0.9,3.1);
  \coordinate (E) at ($(Ap)!0.5!(C)$);
  % 底面 PBCD
  \draw[thick] (P)--(B)--(C)--(D)--cycle;
  \draw[dashed] (A)--(P);
  \draw[dashed] (A)--(D);
  \draw[dashed] (P)--(D);
  % 折起后的 A'
  \draw[thick] (Ap)--(P);
  \draw[thick] (Ap)--(D);
  \draw[thick] (Ap)--(B);
  \draw[thick] (Ap)--(C);
  \draw (D)--(E);
  \fill (E) circle (1.3pt) node[right] {$E$};
  \node[left] at (Ap) {$A'$};
  \node[left] at (A) {$A$};
  \node[left] at (P) {$P$};
  \node[below] at (B) {$B$};
  \node[below] at (C) {$C$};
  \node[above] at (D) {$D$};
\end{tikzpicture}
