\begin{tikzpicture}[>=Stealth, line width=0.8pt, font=\small]
  % 坐标轴
  \draw[->] (-0.95,0) -- (7.25,0) node[above right=-2pt] {$x$};
  \draw[->] (0,-3.75) -- (0,4.05) node[left] {$y$};
  \node[below right] at (0.02,-0.05) {$O$};

  \node[below] at (3.665,0) {$x_0$};
