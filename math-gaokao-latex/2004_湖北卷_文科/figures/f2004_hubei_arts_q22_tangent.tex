\begin{tikzpicture}[scale=1.3, >=stealth, line cap=round, line join=round]
  % 坐标轴
  \draw[->] (-2.5, 0) -- (2.5, 0) node[below] {$x$};
  \draw[->] (0, -0.8) -- (0, 3.5) node[left] {$y$};
  \node[below left] at (0, 0) {$O$};

  % 抛物线 g(x) = x^2 + x + 1
  \draw[thick, smooth, domain=-2.0:1.4, samples=50] 
    plot (\x, {\x*\x + \x + 1}) node[right] {$g(x) = x^2 + bx + c$};

  % 切线 f(x) = x + 1
  \draw[thick, dashed, blue!80!black, domain=-1.8:2.2] 
    plot (\x, {\x + 1}) node[above right] {$f(x) = x + b$};

  % 切点 (0, 1)
  \coordinate (T) at (0, 1);
  \fill[red] (T) circle (1.5pt) node[above left] {$T(0, b)$};
\end{tikzpicture}
