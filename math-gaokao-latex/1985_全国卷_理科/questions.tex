\begin{question}
\gkti{16}设

\nopagebreak[4]
\[
a_n=\sqrt{1\cdot2}+\sqrt{2\cdot3}+\cdots+\sqrt{n(n+1)}
\qquad(n=1,2,\ldots).
\]

\nopagebreak[4]
\[

\Needspace{5\baselineskip}
（2） 设

\nopagebreak[4]
\[
b_n=\frac{a_n}{n(n+1)}
\qquad(n=1,2,\ldots),
\]

\Needspace{5\baselineskip}
用定义证明：

\nopagebreak[4]
\[
\lim_{n\to\infty}b_n=\frac12.
\]
