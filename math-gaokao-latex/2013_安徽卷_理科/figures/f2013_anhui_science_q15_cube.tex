\begin{tikzpicture}[semithick, font=\small]
  \useasboundingbox (-0.6,-0.6) rectangle (3.8,3.15);
  \coordinate (A)  at (0,0);
  \coordinate (B)  at (2.2,0);
  \coordinate (C)  at (3.2,0.62);
