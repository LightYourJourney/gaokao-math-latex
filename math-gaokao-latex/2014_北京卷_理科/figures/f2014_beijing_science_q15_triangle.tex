\begin{tikzpicture}[scale=0.75, line width=0.8pt, font=\small]
  \coordinate (B) at (0,0);
  \coordinate (D) at (3,0);
  \coordinate (C) at (5,0);
  \coordinate (A) at (4,6.9282);   % AB=8，∠B=60°

  \draw (A) -- (B) -- (C) -- cycle;
  \draw (A) -- (D);

  \node[above] at (A) {$A$};
  \node[below left=-1pt] at (B) {$B$};
  \node[below] at (D) {$D$};
  \node[below right=-1pt] at (C) {$C$};
\end{tikzpicture}
