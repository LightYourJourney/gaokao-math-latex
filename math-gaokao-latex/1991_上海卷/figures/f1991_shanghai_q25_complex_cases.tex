% 解析说明图。前三图属于(1)，z=2i允许；第四图属于(2)，omega=2i排除。
% 示例theta=30/90/150度，仅说明推导，不作为新增题设。
\begin{tikzpicture}[x=1cm,y=1cm,line width=0.45pt]
  \foreach \shift/\ang in {0/30,3.2/90,6.4/150}{
    \begin{scope}[xshift=\shift cm]
      \coordinate (O) at (0,0);
      \coordinate (I) at (0,1);
      \coordinate (Z) at ({sin(2*\ang)},{1-cos(2*\ang)});
      \draw[->] (-1.2,0)--(1.3,0) node[right] {$u$};
      \draw[->] (0,-0.25)--(0,2.35) node[above] {$v$};
      \draw (I) circle[radius=1];
      \draw (O)--(Z);
      \draw[dashed] (I)--(Z);
      \draw (0.32,0) arc[start angle=0,end angle=\ang,radius=0.32];
      \ifnum\ang=30
        \node[font=\small] at (0.75,0.17) {$\theta$};
      \else
        \node at ({0.62*cos(\ang/2)},{0.48*sin(\ang/2)}) {$\theta$};
      \fi
      \fill (Z) circle[radius=0.8pt];
      \fill (I) circle[radius=0.7pt];
      \ifnum\ang=150
        \node[right] at (I) {$i$};
      \else
        \node[left] at (I) {$i$};
      \fi
      \node[below left] at (O) {$O$};
      \ifnum\ang=90
        \node[above right] at (Z) {$z=2i$};
      \else
        \ifnum\ang=150
          \node[above left] at (Z) {$z$};
        \else
          \node[above right] at (Z) {$z$};
        \fi
      \fi
      \node at (0,-0.62) {$(1):\ \theta=\ang^\circ$};
    \end{scope}
  }
  \begin{scope}[xshift=9.6cm]
    \draw[->] (-1.2,0)--(1.3,0) node[right] {$u$};
    \draw[->] (0,-0.25)--(0,2.35) node[above] {$v$};
    \draw (0,1) circle[radius=1];
    \fill (0,0) circle[radius=0.9pt];
    \draw[fill=white] (0,2) circle[radius=0.055];
    \node[below left] at (0,0) {$0$};
    \node[above right] at (0,2) {$2i$};
    \node[left] at (0,1) {$i$};
    \node at (0,-0.62) {$(2):\ \omega\text{轨迹}$};
  \end{scope}
\end{tikzpicture}
