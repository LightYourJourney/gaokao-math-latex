\begin{question}
\gkti{19}设复数 $z_1$ 和 $z_2$ 满足关系式

\nopagebreak[4]
\[
z_1\overline{z_2}+\overline A z_1+A\overline{z_2}=0,
\]

其中 $A$ 为不等于 0 的复数。证明：

\Needspace{5\baselineskip}
（1） 证明：

\nopagebreak[4]
\[
|z_1+A||z_2+A|=|A|^2\text{；}
\]

\Needspace{5\baselineskip}
（2） 证明：

\nopagebreak[4]
\[
\frac{z_1+A}{z_2+A}
=\left|
\frac{\overline{z_1}+\overline A}
{\overline{z_2}+\overline A}
\right|\text{。}
\]

\end{question}
