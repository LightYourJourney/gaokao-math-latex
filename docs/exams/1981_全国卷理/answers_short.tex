\gkpart{题目}
\begin{itemize}[leftmargin=2.6em]
  \item[\textbf{1.}] (1) $A\cup B=\mathbf{R}$（全体实数的集合）；(2) $A\cap B=\varnothing$（空集）
  \item[\textbf{2.}] (1) 共 $4\times3=12$ 种。所有结果为（正班长，副班长）：$(A,B)$、$(A,C)$、$(A,D)$、$(B,A)$、$(B,C)$、$(B,D)$、$(C,A)$、$(C,B)$、$(C,D)$、$(D,A)$、$(D,B)$、$(D,C)$；(2) 共 $\mathrm{C}_4^3=4$ 种。所有结果为：$\{A,B,C\}$、$\{A,B,D\}$、$\{A,C,D\}$、$\{B,C,D\}$
  \item[\textbf{3.}] 第一行（$A$：平行四边形；$B$：矩形）为必要条件；第二行（$A$：$a=3$；$B$：$\lvert a\rvert=3$）为充分条件；第三行（$A$：$\theta=150^\circ$；$B$：$\sin\theta=\frac12$）为充分条件；第四行（$A$：点 $(a,b)$ 在圆 $x^2+y^2=R^2$ 上；$B$：$a^2+b^2=R^2$）为充要条件
  \item[\textbf{4.}] $\triangle ABC$ 中 $a^2=b^2+c^2-2bc\cos A$（写出等价的一式即可）
  \item[\textbf{5.}] $x>a+b+c$ 且 $x\ne0$；当 $a+b+c\geqslant0$ 时即 $x>a+b+c$
  \item[\textbf{6.}] 见解析版
  \item[\textbf{7.}] (1) 约 $14.86$ 亿（$14.859$ 亿）；(2) 每年比上年平均递增率最高约为 $0.92\%$
  \item[\textbf{8.}] (1) $\arcsin\dfrac{\sqrt7}{5}$（约 $31^\circ57'$）；(2) $\arcsin\dfrac{\sqrt3}{10}$（约 $9^\circ58'$）
  \item[\textbf{9.}] (1) $2x^2-y^2-4x+y=0$（即 $\dfrac{(x-1)^2}{7/8}-\dfrac{\left(y-\frac12\right)^2}{7/4}=1$，为一条双曲线）；(2) 不存在
\end{itemize}
