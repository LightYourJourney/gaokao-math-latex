% 2015 年陕西卷（文史类）第 5 题：三视图（依固定扫描页重绘，只画拓扑与标号，不按比例）
\begin{tikzpicture}[>=stealth,scale=1.0]
  % 正视图：边长 2 的正方形
  \draw (0,0) rectangle (2,2);
  \draw[<->] (0,-0.3)--(2,-0.3);
  \node at (1,-0.62) {$2$};
  \draw[<->] (2.3,0)--(2.3,2);
  \node at (2.68,1) {$2$};
  \node at (1,-1.05) {正视图};
  % 左视图：宽 1、高 2 的长方形（扫描页未标尺寸）
  \draw (4.0,0) rectangle (5.0,2);
  \node at (4.5,-1.05) {左视图};
  % 俯视图：宽 2、高 1 的半圆
  \draw (0,-2.4)--(2,-2.4);
  \draw (0,-2.4) arc (180:0:1);
  \draw[<->] (2.3,-2.4)--(2.3,-1.4);
  \node at (2.68,-1.9) {$1$};
  \node at (1,-3.9) {俯视图};
\end{tikzpicture}
