\begin{itemize}[leftmargin=2.6em]
  \item[\textbf{19.}] $\dfrac{\sqrt2a}{1-a}$。
  \item[\textbf{20.}] （1）$a=2$；（2）最大值为 $\log_a 49$（当 $x=0$ 时取得）。

  \item[\textbf{23.}] （1）$a=-1$；（2）递减数列；（3）真命题：当 $x_1=\dfrac{1}{1-2^m}$（$m\in\mathbf N^*$）时 $\{x_n\}$ 是有穷数列，且项数为 $m$；这也是充要条件。
\end{itemize}
