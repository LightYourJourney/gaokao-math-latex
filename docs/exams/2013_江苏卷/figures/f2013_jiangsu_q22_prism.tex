% AB⊥AC，AB=AC=2，A1A=4，D 为 BC 中点
% ============================================================
\begin{tikzpicture}[
  line width=0.6pt, font=\small, >=Stealth
]

  \coordinate (A)  at (1.3,0.85);
  \coordinate (B)  at (0,0);
  \coordinate (C)  at (3.4,0);
  \coordinate (A1) at (1.3,3.9);
  \coordinate (B1) at (0,3.05);
  \coordinate (C1) at (3.4,3.05);
  \coordinate (D)  at ($(B)!0.5!(C)$);

  % 底面
  \draw (A) -- (B) -- (C);
  \draw[dashed] (A) -- (C);

  % 侧棱
  \draw (B) -- (B1);
  \draw (C) -- (C1);
  \draw[dashed] (A) -- (A1);

  % 顶面
  \draw (A1) -- (B1) -- (C1) -- cycle;

  % 题设中的线段
  \draw[dashed] (A1) -- (B);
  \draw (C1) -- (D);
  \draw[dashed] (A) -- (D);
  \draw[dashed] (A) -- (C1);

  % 标注
  \node[above left]  at (A)  {$A$};
  \node[below left] at (B)  {$B$};
  \node[right]      at (C)  {$C$};
  \node[left]       at (A1) {$A_1$};
  \node[left]       at (B1) {$B_1$};
  \node[right]      at (C1) {$C_1$};
  \node[below]      at (D)  {$D$};

\end{tikzpicture}
