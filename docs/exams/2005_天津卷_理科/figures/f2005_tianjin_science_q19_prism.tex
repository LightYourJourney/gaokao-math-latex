  % 底面顶点
  \coordinate (A) at (0, 0);
  \coordinate (B) at (2.2, 0);
  \coordinate (C) at (1.8, 0.8);

  % 侧棱向量: (1.2, 2.4)
  \coordinate (A1) at (1.2, 2.4);
  \coordinate (B1) at (3.4, 2.4);

  \node[right] at (B1) {$B_1$};
  \node[above] at (C1) {$C_1$};
