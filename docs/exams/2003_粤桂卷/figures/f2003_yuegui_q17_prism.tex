% 2003 年粤桂卷第 17 题：正四棱柱 ABCD - A1B1C1D1
\begin{tikzpicture}[>=stealth,scale=0.85]
  % Coordinates of vertices
  \coordinate (A) at (0,0);
  \coordinate (B) at (2.2,-0.5);
  \coordinate (C) at (3.4,0.6);
  \coordinate (D) at (1.2,1.1);

  \coordinate (A1) at (0,2.6);
  \coordinate (B1) at (2.2,2.1);
  \coordinate (C1) at (3.4,3.2);
  \coordinate (D1) at (1.2,3.7);

  % Midpoints E (on CC1) and F (on BD1)
  \coordinate (E) at ($(C)!0.5!(C1)$);
  \coordinate (F) at ($(B)!0.5!(D1)$);

  % Back edges (dashed)
  \draw[dashed] (D)--(A) (D)--(C) (D)--(D1);
  \draw[dashed] (B)--(D1);

  % Front edges (solid)
  \draw[thick] (A)--(B)--(C);
  \draw[thick] (A1)--(B1)--(C1)--(D1)--cycle;
  \draw[thick] (A)--(A1) (B)--(B1) (C)--(C1);

  % Segment EF
  \draw[thick] (E)--(F);

  % Node labels
  \fill (A) circle (1.2pt) node[below left] {$A$};
  \fill (B) circle (1.2pt) node[below] {$B$};
  \fill (C) circle (1.2pt) node[right] {$C$};
  \fill (D) circle (1.2pt) node[left] {$D$};
  \fill (A1) circle (1.2pt) node[left] {$A_1$};
  \fill (B1) circle (1.2pt) node[right] {$B_1$};
  \fill (C1) circle (1.2pt) node[above right] {$C_1$};
  \fill (D1) circle (1.2pt) node[above left] {$D_1$};
  \fill (E) circle (1.2pt) node[right] {$E$};
  \fill (F) circle (1.2pt) node[left] {$F$};
\end{tikzpicture}
