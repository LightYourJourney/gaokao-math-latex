  \coordinate (B) at (0,-4.6);
  \coordinate (C) at (1.9,1.9);
  \draw[-stealth] (0,1.5) -- (0,2.3);
  \node[above right] at (0,2.15) {北};

  \fill (A) circle (1.2pt) node[below left] {$A$};
