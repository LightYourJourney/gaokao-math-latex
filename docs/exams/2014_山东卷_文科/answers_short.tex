\gkpart{填空题}
\begin{itemize}[leftmargin=2.6em]
  \item[\textbf{11.}] $3$。$x=1,2,3$ 时条件成立，依次得 $n=1,2,3$；$x=4$ 时 $3>0$ 不成立，输出 $n=3$。
  \item[\textbf{12.}] $\pi$。$y=\sin\left(2x+\dfrac\pi6\right)+\dfrac12$，最小正周期为 $\dfrac{2\pi}2=\pi$。
  \item[\textbf{13.}] $12$。底面积 $6\sqrt3$，由 $V=2\sqrt3$ 得高 $h=1$；底面边心距 $\sqrt3$，斜高 $2$，侧面积 $6\times\dfrac12\times2\times2=12$。
  \item[\textbf{14.}] $(x-2)^2+(y-1)^2=4$。设圆心 $(2t,t)$、半径 $2t$，由 $4t^2=t^2+3$ 得 $t=1$。
  \item[\textbf{15.}] $y=\pm x$。由 $|FA|=c$ 得 $\dfrac{p^2}4=b^2$，再由准线截得弦长 $2c$ 得 $c^2=2a^2$，故 $b=a$。
\end{itemize}

\gkpart{解答题}
\begin{itemize}[leftmargin=2.6em]
  \item[\textbf{16.}] (1) $A$ 地区 $1$ 件，$B$ 地区 $3$ 件，$C$ 地区 $2$ 件。三层数量之比为 $1:3:2$，共抽 $6$ 件；(2) $\dfrac4{15}$。$\dfrac{\mathrm C_1^2+\mathrm C_3^2+\mathrm C_2^2}{\mathrm C_6^2}=\dfrac{0+3+1}{15}=\dfrac4{15}$
  \item[\textbf{17.}] (1) $b=3\sqrt2$。$\sin B=\cos A=\dfrac{\sqrt6}3$，$\sin A=\dfrac{\sqrt3}3$，由正弦定理得 $b=3\sqrt2$；(2) $\dfrac{3\sqrt2}2$。$C=\dfrac\pi2-2A$，$\sin C=\cos2A=\dfrac13$，故 $S=\dfrac12ab\sin C=\dfrac{3\sqrt2}2$
  \item[\textbf{18.}] (1) 见解析版；(2) 见解析版
  \item[\textbf{19.}] (1) $a_n=2n$。由 $a_2^2=a_1a_4$ 得 $a_1=2$；(2) $b_n=n(n+1)$；$T_n=\dfrac{n(n+2)}2$（$n$ 偶），$T_n=-\dfrac{(n+1)^2}2$（$n$ 奇）
  \item[\textbf{20.}] (1) $y=\dfrac12x-\dfrac12$。$f(1)=0$，$f'(1)=\dfrac12$；(2) $a\geqslant0$ 时在 $(0,+\infty)$ 上递增；$a\leqslant-\dfrac12$ 时在 $(0,+\infty)$ 上递减；$-\dfrac12<a<0$ 时在两根外递减、两根间递增（两根见解析版）
  \item[\textbf{21.}] (1) $\dfrac{x^2}4+y^2=1$。由 $e=\dfrac{\sqrt3}2$ 得 $b^2=\dfrac{a^2}4$，再由弦长 $\dfrac{4\sqrt{10}}5$ 得 $a^2=4$；(2) ① $\lambda=-\dfrac12$；② $\dfrac98$。设 $A(x_1,y_1)$，由 $AD\perp AB$ 与点在椭圆上得 $k_1=\dfrac{y_1}{4x_1}$，进而 $k_2=-2k_1$；面积 $S=\dfrac98|x_1y_1|\leqslant\dfrac98$
\end{itemize}
