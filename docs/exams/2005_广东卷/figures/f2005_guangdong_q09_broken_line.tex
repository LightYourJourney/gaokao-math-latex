\begin{tikzpicture}[>=stealth, scale=1.0]
  % 坐标轴
  \draw[->, thick] (-2.8, 0) -- (2.8, 0) node[below] {$x$};
  \draw[->, thick] (0, -0.6) -- (0, 3.8) node[left] {$y$};
  \node[below left] at (0, 0) {$O$};

  % 刻度
  \foreach \x in {-2, -1, 1, 2} {
    \draw (\x, 0) -- (\x, 0.1);
    \node[below] at (\x, 0) {$\x$};
  }
  \foreach \y in {1, 2, 3} {
    \draw (0, \y) -- (0.1, \y);
    \node[left] at (0, \y) {$\y$};
  }

  % 虚线辅助线
  \draw[dashed] (1, 0) -- (1, 3) -- (0, 3);

  % 折线
  \draw[very thick] (-2, 0) -- (0, 1) -- (1, 3);
