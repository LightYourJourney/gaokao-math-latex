```python
# tools/schema_validate.py 的核心接口
from __future__ import annotations

import re
from typing import Any, Mapping, Sequence

SUPPORTED = {
    "$schema", "$id", "$defs", "$ref", "type", "required", "properties",
    "items", "enum", "const", "pattern", "minimum", "minItems",
    "uniqueItems", "additionalProperties", "oneOf", "allOf",
    "description", "title", "default",
}


class SchemaError(ValueError):
    pass


def validate(instance: Any, schema: Mapping[str, Any]) -> None:
    _validate(instance, schema, schema, "$")


def _resolve_ref(root: Mapping[str, Any], ref: str) -> Mapping[str, Any]:
    if not ref.startswith("#/"):
        raise SchemaError(f"只支持本地 $ref：{ref}")
    node: Any = root
    for part in ref[2:].split("/"):
        node = node[part.replace("~1", "/").replace("~0", "~")]
    if not isinstance(node, Mapping):
        raise SchemaError(f"$ref 目标不是对象：{ref}")
    return node


def _validate(value: Any, schema: Mapping[str, Any], root: Mapping[str, Any], path: str) -> None:
    unknown = set(schema) - SUPPORTED
    if unknown:
        raise SchemaError(f"{path}: 不支持的 Schema 关键字：{sorted(unknown)}")
    if "$ref" in schema:
        _validate(value, _resolve_ref(root, schema["$ref"]), root, path)
        return
    for branch in schema.get("allOf", []):
        _validate(value, branch, root, path)
    if "oneOf" in schema:
        passed = 0
        for branch in schema["oneOf"]:
            try:
                _validate(value, branch, root, path)
                passed += 1
            except SchemaError:
                pass
        if passed != 1:
            raise SchemaError(f"{path}: oneOf 应恰好命中一项，实际 {passed}")
    expected = schema.get("type")
    type_map = {
        "object": dict, "array": list, "string": str,
        "integer": int, "number": (int, float), "boolean": bool,
        "null": type(None),
    }
    if expected is not None:
        allowed = expected if isinstance(expected, list) else [expected]
        if not any(isinstance(value, type_map[item]) and not (
            item in {"integer", "number"} and isinstance(value, bool)
        ) for item in allowed):
            raise SchemaError(f"{path}: 类型应为 {allowed}")
    if "enum" in schema and value not in schema["enum"]:
        raise SchemaError(f"{path}: 值不在 enum 中")
    if "const" in schema and value != schema["const"]:
        raise SchemaError(f"{path}: 值必须等于 {schema['const']!r}")
    if isinstance(value, str) and "pattern" in schema and not re.search(schema["pattern"], value):
        raise SchemaError(f"{path}: 字符串不匹配 {schema['pattern']}")
    if isinstance(value, (int, float)) and "minimum" in schema and value < schema["minimum"]:
        raise SchemaError(f"{path}: 数值小于 minimum")
    if isinstance(value, list):
        if len(value) < schema.get("minItems", 0):
            raise SchemaError(f"{path}: 数组长度不足")
        if schema.get("uniqueItems") and len({repr(item) for item in value}) != len(value):
            raise SchemaError(f"{path}: 数组元素不唯一")
        if "items" in schema:
            for index, item in enumerate(value):
                _validate(item, schema["items"], root, f"{path}[{index}]")
    if isinstance(value, dict):
        for name in schema.get("required", []):
            if name not in value:
                raise SchemaError(f"{path}.{name}: 缺少必填字段")
        properties = schema.get("properties", {})
        if schema.get("additionalProperties") is False:
            extra = set(value) - set(properties)
            if extra:
                name = sorted(extra)[0]
                raise SchemaError(f"{path}.{name}: 未知字段")
        for name, child in value.items():
            if name in properties:
                _validate(child, properties[name], root, f"{path}.{name}")
```

```python
# tools/manifest.py 的核心接口
from __future__ import annotations

import hashlib
import json
import os
import tempfile
from pathlib import Path
from typing import Any, Mapping


def canonical_json(value: Any) -> str:
    return json.dumps(value, ensure_ascii=False, indent=2, sort_keys=True) + "\n"


def atomic_write_json(path: Path, value: Any) -> None:
    path.parent.mkdir(parents=True, exist_ok=True)
    fd, raw = tempfile.mkstemp(prefix=f".{path.name}.", dir=path.parent)
    tmp = Path(raw)
    try:
        with os.fdopen(fd, "w", encoding="utf-8", newline="\n") as handle:
            handle.write(canonical_json(value))
            handle.flush()
            os.fsync(handle.fileno())
        os.replace(tmp, path)
    finally:
        if tmp.exists():
            tmp.unlink()


def file_sha256(path: Path) -> str:
    digest = hashlib.sha256()
    with path.open("rb") as handle:
        for chunk in iter(lambda: handle.read(1024 * 1024), b""):
            digest.update(chunk)
    return digest.hexdigest()


def validate_manifest_semantics(manifest: Mapping[str, Any]) -> None:
    source_ids = [item["source_id"] for item in manifest["sources"]]
    if len(source_ids) != len(set(source_ids)):
        raise ValueError("source_id 不唯一")
    paper_ids = [item["paper_id"] for item in manifest["papers"]]
    if len(paper_ids) != len(set(paper_ids)):
        raise ValueError("paper_id 不唯一")
    known_papers = set(paper_ids)
    for source in manifest["sources"]:
        relation = source.get("paper_relation")
        paper_id = source.get("paper_id")
        if relation is not None and paper_id is None:
            raise ValueError(f"{source['source_id']}: paper_relation 存在但 paper_id 为空")
        if paper_id is not None and paper_id not in known_papers:
            raise ValueError(f"{source['source_id']}: paper_id 不存在")
        if source["source_state"] == "non_paper" and (relation is not None or paper_id is not None):
            raise ValueError(f"{source['source_id']}: non_paper 不得绑定试卷")
```

```python
# tools/source_manifest.py 的 inventory 核心
from __future__ import annotations

import argparse
import hashlib
import subprocess
from pathlib import Path
from typing import Any

UPSTREAM_URL = "https://github.com/deekur/gaokaomath"
BASELINE_COMMIT = "34c13a6cf632c0b69c95f89b9b923bbc4a08924d"
ROOTS = ("普通高考", "春季高考")


def parse_ls_tree(raw: bytes) -> list[dict[str, str]]:
    result = []
    for row in raw.split(b"\0"):
        if not row:
            continue
        meta, path_raw = row.split(b"\t", 1)
        mode, kind, oid = meta.split()
        if kind == b"blob":
            result.append({
                "mode": mode.decode("ascii"),
                "oid": oid.decode("ascii"),
                "path": path_raw.decode("utf-8", "strict"),
            })
    return result


def source_id(url: str, commit: str, path: str) -> str:
    payload = b"source-v1\0" + url.encode() + b"\0" + commit.encode() + b"\0" + path.encode("utf-8")
    return "src-gaokaomath-v1-" + hashlib.sha256(payload).hexdigest()


def git_tree(repo: Path, commit: str) -> list[dict[str, str]]:
    head = subprocess.run(
        ["git", "-C", str(repo), "rev-parse", "HEAD"],
        check=True, capture_output=True, text=True,
    ).stdout.strip()
    if head != commit:
        raise RuntimeError(f"上游 HEAD {head} 与固定 commit {commit} 不一致")
    completed = subprocess.run(
        ["git", "-C", str(repo), "ls-tree", "-r", "-z", "--full-tree", commit, "--", *ROOTS],
        check=True, capture_output=True,
    )
    return parse_ls_tree(completed.stdout)


def build_inventory(entries: list[dict[str, str]], url: str, commit: str) -> dict[str, Any]:
    pdfs = [item for item in entries if item["path"].casefold().endswith(".pdf")]
    excluded = sorted(item["path"] for item in entries if not item["path"].casefold().endswith(".pdf"))
    sources = []
    counts = {"regular": 0, "spring": 0}
    for item in sorted(pdfs, key=lambda row: row["path"].encode("utf-8")):
        exam_type = "regular" if item["path"].startswith("普通高考/") else "spring"
        counts[exam_type] += 1
        sources.append({
            "source_id": source_id(url, commit, item["path"]),
            "upstream_path": item["path"],
            "git_blob_oid": item["oid"],
            "source_state": "not_started",
            "paper_relation": None,
            "paper_id": None,
            "artifact": None,
            "issue_ids": [],
        })
    return {
        "$schema": "./schemas/source-manifest.schema.json",
        "schema_version": 1,
        "upstream": {"repository_url": url, "commit": commit, "scoped_roots": list(ROOTS)},
        "inventory": {
            "counts": {"total_pdf": len(sources), "by_exam_type": counts},
            "excluded_paths": excluded,
        },
        "sources": sources,
        "papers": [],
        "batches": [],
    }
```

实现必须按以下顺序：`git cat-file blob <oid>` 写同目录临时文件；`git hash-object` 验证 OID；检查 `%PDF-`；计算 SHA-256；运行 `pdfinfo` 并解析 `Pages:`；`os.replace` 到上游工作树相对路径；最后才原子更新 manifest。失败不得覆盖旧文件或旧 artifact。

```python
def inspect_pdf(path: Path, pdfinfo_bin: str) -> dict[str, Any]:
    if not path.read_bytes()[:5] == b"%PDF-":
        raise RuntimeError("文件头不是 PDF")
    completed = subprocess.run(
        [pdfinfo_bin, str(path)], capture_output=True, text=True,
    )
    if completed.returncode != 0:
        raise RuntimeError(f"pdfinfo 校验失败：{completed.stderr.strip()}")
    pages = None
    for line in completed.stdout.splitlines():
        if line.startswith("Pages:"):
            pages = int(line.split(":", 1)[1].strip())
    if pages is None or pages < 1:
        raise RuntimeError("pdfinfo 未返回正页数")
    return {"page_count": pages, "pdf_check": "passed", "header_is_pdf": True}
```

```python
    referenced: list[str] = []
    # 复用现有块解析逻辑；所有诊断通过局部列表返回。
    text = convert_block(strip_frontmatter(source_text.splitlines()), context, warnings, referenced)
    if context.strict and warnings:
        raise ValueError("\n".join(warnings))
    return ConversionResult("\n".join(text) + "\n", tuple(warnings), tuple(referenced))
```
