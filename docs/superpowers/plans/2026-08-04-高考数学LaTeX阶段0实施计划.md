```python
# tools/schema_validate.py 的核心接口
from __future__ import annotations

import re
from typing import Any, Mapping, Sequence

SUPPORTED = {
    "$schema", "$id", "$defs", "$ref", "type", "required", "properties",
    "items", "enum", "const", "pattern", "minimum", "minItems",
    "uniqueItems", "additionalProperties", "oneOf", "allOf",
    "description", "title", "default",
}


class SchemaError(ValueError):
    pass


def validate(instance: Any, schema: Mapping[str, Any]) -> None:
    _validate(instance, schema, schema, "$")


def _resolve_ref(root: Mapping[str, Any], ref: str) -> Mapping[str, Any]:
    if not ref.startswith("#/"):
        raise SchemaError(f"只支持本地 $ref：{ref}")
    node: Any = root
    for part in ref[2:].split("/"):
        node = node[part.replace("~1", "/").replace("~0", "~")]
    if not isinstance(node, Mapping):
        raise SchemaError(f"$ref 目标不是对象：{ref}")
    return node


def _validate(value: Any, schema: Mapping[str, Any], root: Mapping[str, Any], path: str) -> None:
    unknown = set(schema) - SUPPORTED
    if unknown:
        raise SchemaError(f"{path}: 不支持的 Schema 关键字：{sorted(unknown)}")
    if "$ref" in schema:
        _validate(value, _resolve_ref(root, schema["$ref"]), root, path)
        return
    for branch in schema.get("allOf", []):
        _validate(value, branch, root, path)
    if "oneOf" in schema:
        passed = 0
        for branch in schema["oneOf"]:
            try:
                _validate(value, branch, root, path)
                passed += 1
            except SchemaError:
                pass
        if passed != 1:
            raise SchemaError(f"{path}: oneOf 应恰好命中一项，实际 {passed}")
    expected = schema.get("type")
    type_map = {
        "object": dict, "array": list, "string": str,
        "integer": int, "number": (int, float), "boolean": bool,
        "null": type(None),
    }
    if expected is not None:
        allowed = expected if isinstance(expected, list) else [expected]
        if not any(isinstance(value, type_map[item]) and not (
            item in {"integer", "number"} and isinstance(value, bool)
        ) for item in allowed):
            raise SchemaError(f"{path}: 类型应为 {allowed}")
    if "enum" in schema and value not in schema["enum"]:
        raise SchemaError(f"{path}: 值不在 enum 中")
    if "const" in schema and value != schema["const"]:
        raise SchemaError(f"{path}: 值必须等于 {schema['const']!r}")
    if isinstance(value, str) and "pattern" in schema and not re.search(schema["pattern"], value):
        raise SchemaError(f"{path}: 字符串不匹配 {schema['pattern']}")
    if isinstance(value, (int, float)) and "minimum" in schema and value < schema["minimum"]:
        raise SchemaError(f"{path}: 数值小于 minimum")
    if isinstance(value, list):
        if len(value) < schema.get("minItems", 0):
            raise SchemaError(f"{path}: 数组长度不足")
        if schema.get("uniqueItems") and len({repr(item) for item in value}) != len(value):
            raise SchemaError(f"{path}: 数组元素不唯一")
        if "items" in schema:
            for index, item in enumerate(value):
                _validate(item, schema["items"], root, f"{path}[{index}]")
    if isinstance(value, dict):
        for name in schema.get("required", []):
            if name not in value:
                raise SchemaError(f"{path}.{name}: 缺少必填字段")
        properties = schema.get("properties", {})
        if schema.get("additionalProperties") is False:
            extra = set(value) - set(properties)
            if extra:
                name = sorted(extra)[0]
                raise SchemaError(f"{path}.{name}: 未知字段")
        for name, child in value.items():
            if name in properties:
                _validate(child, properties[name], root, f"{path}.{name}")
```

```python
# tools/manifest.py 的核心接口
from __future__ import annotations

import fcntl
import hashlib
import json
import os
import tempfile
from pathlib import Path
from typing import Any, Mapping


def durable_fsync(fd: int) -> str:
    full = getattr(fcntl, "F_FULLFSYNC", None)
    if full is not None:
        try:
            fcntl.fcntl(fd, full)
            return "fullfsync"
        except OSError:
            pass
    os.fsync(fd)
    return "fsync"


def fsync_directory(path: Path) -> None:
    fd = os.open(str(path), os.O_RDONLY)
    try:
        os.fsync(fd)
    finally:
        os.close(fd)


def ensure_durable_directory(path: Path) -> None:
    missing = []
    current = path
    while not current.exists():
        missing.append(current)
        current = current.parent
    for directory in reversed(missing):
        directory.mkdir()
        fsync_directory(directory)
        fsync_directory(directory.parent)


def canonical_json(value: Any) -> str:
    return json.dumps(value, ensure_ascii=False, indent=2, sort_keys=True) + "\n"


def atomic_write_json(path: Path, value: Any) -> None:
    ensure_durable_directory(path.parent)
    fd, raw = tempfile.mkstemp(prefix=f".{path.name}.", dir=path.parent)
    tmp = Path(raw)
    try:
        with os.fdopen(fd, "w", encoding="utf-8", newline="\n") as handle:
            handle.write(canonical_json(value))
            handle.flush()
            durable_fsync(handle.fileno())
        os.replace(tmp, path)
        fsync_directory(path.parent)
    finally:
        if tmp.exists():
            tmp.unlink()


def file_sha256(path: Path) -> str:
    digest = hashlib.sha256()
    with path.open("rb") as handle:
        for chunk in iter(lambda: handle.read(1024 * 1024), b""):
            digest.update(chunk)
    return digest.hexdigest()


def validate_manifest_semantics(manifest: Mapping[str, Any]) -> None:
    source_ids = [item["source_id"] for item in manifest["sources"]]
    if len(source_ids) != len(set(source_ids)):
        raise ValueError("source_id 不唯一")
    paper_ids = [item["paper_id"] for item in manifest["papers"]]
    if len(paper_ids) != len(set(paper_ids)):
        raise ValueError("paper_id 不唯一")
    known_papers = set(paper_ids)
    for source in manifest["sources"]:
        relation = source.get("paper_relation")
        paper_id = source.get("paper_id")
        if relation is not None and paper_id is None:
            raise ValueError(f"{source['source_id']}: paper_relation 存在但 paper_id 为空")
        if paper_id is not None and paper_id not in known_papers:
            raise ValueError(f"{source['source_id']}: paper_id 不存在")
        if source["source_state"] == "non_paper" and (relation is not None or paper_id is not None):
            raise ValueError(f"{source['source_id']}: non_paper 不得绑定试卷")
    for paper in manifest["papers"]:
        reverse = sorted(
            source["source_id"] for source in manifest["sources"]
            if source.get("paper_id") == paper["paper_id"]
            and source.get("paper_relation") is not None
        )
        if sorted(paper["source_ids"]) != reverse:
            raise ValueError(f"{paper['paper_id']}: source_ids 双向关系不一致")
```

每条 batch 写 `batch_id/paper_ids/status=planned/issues=[]`。batch 工作流状态枚举为 `planned/provenance_verified/compiled_pending_review/closed/closed_with_blockers`，只能由 `reconcile_batch_status()` 直接检查 provenance、生成物、build、review 和 issues 当前事实后在 CAS 事务中写入，不允许从 paper 缓存状态反推，也不提供任意 setter：

```text
任一卷 provenance 缺失/失效                              → planned
否则任一卷 build 缺失/running/failed/stale                → provenance_verified
否则任一卷 build=compiled_pending_review                  → compiled_pending_review
否则任一卷 build=succeeded 且 review 缺失/pending/failed/stale
                                                           → compiled_pending_review
否则所有 build=succeeded/review=passed 但 batch report 缺失/stale
                                                           → compiled_pending_review
否则全部 derive_status()==complete                        → closed
否则所有卷均为 complete|blocked 且至少一卷 blocked         → closed_with_blockers
```

只有所有卷 build receipt 都当前且状态恰为 `compiled_pending_review` 时，`prepare-batch` 才能开始视觉审查；不得只信缓存 batch status。build `running` 时无论旧 review 状态为何都留在 `provenance_verified`。

该优先级覆盖混合状态且唯一；无法命中任何分支视为状态机缺陷并失败，不猜测。paper `production_state` 由同一事务同步为上述映射结果，`check-all` 现场重算并拒绝缓存漂移。

因此合法恢复包括 `closed_with_blockers → closed`（问题解决）、任一 `closed* → provenance_verified`（构建输入变化但 provenance 仍当前）以及 `closed* → planned`（provenance 失效）；每次都记录旧/新状态和触发 fingerprint 到 transaction journal/报告，不静默改写历史。Tasks 17–22 的 `import-batch`、Task 23 formal build、Tasks 25–30 `finalize-batch` 以及后续问题解决都调用同一个 reconciler。

Git 持久化是独立只读派生 `git_state=validated|committed`：finalize 事务完成、所有内容门禁当前但 ownership paths 相对 HEAD 有变化时为 `validated`；这些精确路径已包含在 HEAD、工作树/索引对其均零 diff 且能由 `git log -- <paths>` 定位提交时为 `committed`。manifest/批次报告不保存包含自身的 commit hash；`provenance.py status` 和 Task 31 `report.py check` 动态重算，最终要求六批均 `committed`。后续任务只处理当前批次，批内最多四套。

- [ ] **Step 1: 写 Schema、答案哈希、枚举、角色、U 级和 blocker 测试**

每题要求 `question_id/printed_number/source_ref/source_sha256/source_pages/question_text_sha256/answer_sha256/transcription_review/answer_evidence/solver_role/reviewer_role/conflict_status/issue_ids`。`transcription_review` 是含 `role/result/evidence` 的对象；`source_ref` 用 oneOf 支持上游 source_id 或本地 legacy 相对路径+SHA-256。上游引用必须属于当前 paper 的 `source_ids`，对应 source 必须 `ready`、artifact 当前且双向关系成立；`source_pages` 必须为去重升序整数并全部位于 `1..artifact.page_count`。本地 legacy 只允许 paper 显式登记的 `legacy_source_files`，现场重算 SHA，且仅用于没有确认上游 source 的既有回忆版；身份 blocker 继续阻止 complete。题面、答案或来源任一哈希变化都使核验失效。测试必须含“引用另一卷的有效 source”“source 非 ready”“页码越界”和“本地路径不属于本卷”四个假绿反例。枚举冻结为：`transcription_review.result=passed|failed|uncertain`，`answer_evidence.grade=A|B|C|D|U`，`conflict_status=none|resolved|unresolved`；`transcription_review.role/solver_role/reviewer_role` 非空且两两不同。测试必须证明 failed/uncertain transcription、U、unresolved conflict、角色重复、review pending/failed、issue 为 open/investigating/deferred 时都不能 complete。

`question_text_sha256` 固定为 `_试题.md` 中该题题面块去除行尾空白、统一 LF 后的 UTF-8 SHA-256。`answer_sha256` 不对自由文本直接取 hash：必须复用 Task 14 `parse_answer_points()` 生成 `AnswerRecord`，按 `question_id` 与小问序号 UTF-8 码点排序，构造 `{"question_id":...,"parts":[{"part":...,"answer":...}]}`，用 Task 1 `canonical_json()` 序列化后取 UTF-8 SHA-256。`source_sha256` 从 manifest artifact 或本地 legacy 文件现场重算。

- [ ] **Step 2: 实现显式完成判定和三角色 `.work` 导入**

`load_provenance_facts()` 显式消费 manifest、机器解析的问题总账和当前生成物，不搜索自然语言；它输出 Task 15 status 核心所需的闭合 facts。status 核心只有在逐题题面核对全部 `passed`、答案全部非 U、冲突全部 `none|resolved`、三角色两两不同、没有 `blocks_completion(issue)`、生成物当前、build/review currency 当前、build 为 `succeeded`、review 为 `passed` 且最终 batch report 同事务验证通过时才返回 `complete`；最终回执通过但仍有 blocker 返回 `blocked`，其他状态按 Task 4 fail-closed 映射。不存在人工 `--set-complete`。`import_batch()` 在 `state` 事务中写 provenance/manifest/issues；formal build 用同一 facts adapter 写 running/结果 receipts 和状态缓存；`check-all` 现场重算并拒绝任何缓存漂移。

`init-batch` 为每套创建三个独占路径：

```text
.work/<batch>/provenance/question-review/<paper>.json
.work/<batch>/provenance/answer-solve/<paper>.json
.work/<batch>/provenance/answer-review/<paper>.json
```

题面核对、独立求解、答案盲核角色分别只写自己的文件，角色身份必须不同；任何角色不得读取答案复核文件。`import-batch` 要求三组 paper 集合恰好等于 batch、题号一致；任一 `transcription_review=failed|uncertain`、答案 grade U 或 `conflict_status=unresolved` 都必须关联 fingerprint 当前、`impact=blocks_completion/status!=resolved` 的 issue，否则拒绝导入。主流程再从当前 tracked `_试题.md`、`_答案要点.md` 和 source artifact 独立重算三种 hash；work 文件中的 hash 只是待比较声明。然后使用 Task 1 `Transaction.prepare(mode="state", ..., expected_revision=...)` 一次提交正式 provenance、新 manifest 和新 issues，manifest 最后由 reconciler 推进 batch 到 `provenance_verified`。失败或崩溃由 transaction journal 恢复，不能留下半提交。

  - `output-paths --manifest PATH --batch ID [--include-state] [--report PATH] --nul`；`check-staged --pathspec-file PATH --nul`。
  parser、work/report containment、选择器和退出码由测试冻结。`finalize-batch` 是唯一领域写入口；`finalize-batch-checked` 是 Tasks 25–30 的受测主流程包装：进入时要求 tracked 工作树/索引干净，调用同一领域函数；0/`FINALIZED` 原样返回且不 stage，4/`VISUAL_FAILED` 时使用不含 report 的 ownership paths 精确 `git add`、`check-staged`、`git diff --cached --check` 并提交固定诊断消息后返回 4，2/6 或其他结果零 stage/零 commit。它用参数数组调用 Git，禁止 shell=True。旧的独立 `aggregate/finalize/close/report` 写命令不存在。`output-paths --include-state --report ... --nul` 输出当前批次所有权范围。`check-staged` 计算该范围内相对 HEAD 实际变更/新增/删除集合，要求这些变化全已 staged 且拒绝范围外 staged 路径；不要求字节未变文件出现在 staged diff。测试覆盖 issues 不变与新增/resolve 两种路径。所有门禁支持 `--json`：0=`FINALIZED`，2=usage，4=`VISUAL_FAILED`（已事务保存诊断状态），6=`INVALID_OR_STALE`（零修改）。

- [ ] **Step 1: 写 stale、缺版本、全页覆盖、视觉失败和 Overfull 豁免测试**

review 状态枚举为 `pending/passed/failed`。每个版本必须按 build receipt 的 `page_count` 精确提交页集合 `1..N`（无缺页、重页或越界），每页都有 `page/result=pass|fail/checks`，统一 checks 至少覆盖字形、标题/题号、选项/公式对齐、图形与标注、页眉页脚、分页/空白/重复、字号和答案泄漏；不适用项必须显式 `not_applicable` 并给理由。review 必须绑定稳定 build input projection/tree、build/toolchain fingerprint、视觉工具 pdftotext/pdftoppm fingerprint 和各 PDF SHA；任一输入或工具链变化即 stale。任一页 fail、四版本不全或页集合不精确，正式 review 都必须为 `failed`。Overfull 豁免要求宽度、源行、PDF 页、visual_result 和理由；警告集合变化时旧豁免失效。

- [ ] **Step 2: 实现诚实的候选页启发式**

`text_chars` 为 pdftotext 每页非空白字符数；`math_glyph_proxy` 为 `=+-×÷∑∫√≤≥∈` 等符号计数；`ink_ratio` 为 pdftoppm 灰度图非白像素比。三个指标在同一 PDF 内 min-max 归一化后等权平均，最高分为候选。算法不宣称理解公式或 TikZ；它只标记重点二次复核页，不能缩小全页审查集合。人工改选记录原候选、改选页和理由。manifest `figure_files` 与 Markdown 引用的精确一致性由 Task 13 结构门禁负责，实际渲染效果由全页视觉审查覆盖。

- [ ] **Step 3: 实现批次工作项和单写聚合**

`prepare-batch` 逐卷直接要求 build receipt 当前且状态恰为 `compiled_pending_review`，不只信缓存 batch status；为 2–4 套生成四版本 work JSON，四个版本代理只写 `.work/<batch>/review/<edition>/<paper>.json`。

`finalize-batch` 在内存完成全部闭环后只提交一次 `state` 事务：
1. 验证 paper/edition 集合完整、每份 PDF 页集合精确为 `1..page_count`、稳定 input tree/toolchain/PDF SHA 与 build 一致，产生 review 候选；
2. 按 `origin=visual-review/paper_id/edition/build_attempt_id/pdf_sha256` 做当前 fail 集合与旧视觉问题集合对账，生成新 issues 候选；
3. 只有四版本 review 全部 `passed`、无未解决视觉 issue、警告豁免逐项匹配时，纯 `finalize_receipt()` 才产生 build `succeeded` 候选；review fail 时 build 保持 `compiled_pending_review`；
4. 用候选 receipts/issues 调用 `derive_content_outcome(facts_without_report)`，只得到 `eligible_complete|eligible_blocked` 内容候选供报告渲染，不生成可持久化 complete/closed；在内存渲染并逐字段验证最终 batch report；验证通过后设置 `batch_report_valid=true`，再唯一一次调用 `derive_status()` 与 `reconcile_batch_status()` 生成最终 paper/batch 状态；
5. 成功路径一次事务提交 review receipts、final build receipts、issues、最终 batch report、paper/batch 状态，manifest 最后写。结构不完整、currency/CAS 不匹配或报告渲染/校验失败时零正式修改。

视觉检查本身出现 fail 是有效负结果：`finalize-batch` 以单独 `state` 事务只提交 `review=failed`、视觉 issues、paper=`blocked` 与 batch=`compiled_pending_review`（build 保持 `compiled_pending_review`），不生成/覆盖最终 batch report，并以退出码 4、JSON `code=VISUAL_FAILED` 停止。执行者用同一 ownership-path 门禁精确提交该诊断状态后，才按恢复流程修源、重编和重审；禁止带未提交诊断状态继续修改规范源。

所有内容无 blocker 时 batch 为 `closed`；任一卷仍有 U/identity/未 resolved completion blocker 时为 `closed_with_blockers`，但两者都要求 build `succeeded`、review `passed` 和 report 候选验证通过。`finalize-batch` 还支持 report-only reconciliation：当现有 build/review 均 current 且为 succeeded/passed、只有 blocker/provenance/report facts 变化时，复用并重新验证现有 receipts，不要求重新 `prepare-batch`，重渲染报告后合法完成 `closed_with_blockers → closed`；若任何 build/review currency 变化则拒绝并要求重编/重审。视觉问题集合对账沿用上文规则：当前 fail upsert，旧 attempt/PDF 中已不在当前 fail 集合的自动视觉问题 resolve；人工/来源问题绝不自动解决。测试覆盖失败页删除、图形跨页、warning 行号/宽度变化、报告渲染失败零修改和事务中断恢复。

视觉 fail 的恢复由受测 `finalize-batch-checked` 执行：它只在领域结果严格为 4/`VISUAL_FAILED` 时提交不含最终报告的诊断 ownership changes 并返回 4，普通 `set -e` 随即停止；2/6 或其他结果零提交停止。之后修唯一规范源/模板/公共层 → 运行 Tasks 13–14 生成器 `--check`（有改动则 `--write` 后全量回归）→ `build_projects.py build --formal --batch <id> --issues ...` → 重新 `prepare-batch` 和全页审查。禁止直接编辑 review/build receipt 或仅把 issue 改成 resolved。该包装器的成功/视觉失败/usage/invalid/CAS 分支都有 CLI 集成测试。
